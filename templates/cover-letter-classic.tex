% Disable fi/fl/ffi/ffl ligatures: PDF text extractors (what ATS systems read)
% decode the ligature glyphs back to U+FB01/FB02/FB03, so a word like
% "verification" becomes a ligature form and a literal keyword search misses it.
%   - pdftex (pdflatex): \DisableLigatures works.
%   - XeTeX (tectonic): fontspec with Ligatures=NoCommon on the bundled
%     Latin Modern font. The ATS extraction preflight also catches any
%     surviving ligature glyphs as a hard blocker.
\ifdefined\pdftexversion
  \DisableLigatures{encoding=T1,family=*}
\else
  \usepackage{fontspec}
  \setmainfont{lmroman10-regular.otf}[Ligatures=NoCommon]
\fi
